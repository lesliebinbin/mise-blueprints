#+title: React Copier Template

This branch is a standalone Copier template for a mise-managed React and Vite
TypeScript application.

* Generate an application

#+begin_src shell
copier copy \
  --vcs-ref react/v0.0.1 \
  git@github.com:lesliebinbin/mise-blueprints.git \
  ./my-react-app
#+end_src

Use =--defaults= to accept every default answer non-interactively:

#+begin_src shell
copier copy \
  --defaults \
  --vcs-ref react/v0.0.1 \
  git@github.com:lesliebinbin/mise-blueprints.git \
  ./my-react-app
#+end_src

* Pinning the template

=--vcs-ref= accepts one Git branch, tag, or commit SHA. For repeatable
automation, use an immutable tag or full commit SHA after the branch is
finalized:

#+begin_src shell
copier copy \
  --vcs-ref <tag-or-full-commit-sha> \
  git@github.com:lesliebinbin/mise-blueprints.git \
  ./my-react-app
#+end_src

* Local development

#+begin_src shell
copier copy \
  --data-file answers.yml.example \
  . \
  /tmp/example-react-app
#+end_src

* Generated application setup

#+begin_src shell
cd my-react-app
mise install
mise setup
mise check
#+end_src
