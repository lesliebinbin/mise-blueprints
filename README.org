#+title: uv Python Copier Template

This branch is a standalone Copier template for a mise- and uv-managed Python
package.

* Generate a package

#+begin_src shell
copier copy \
  --vcs-ref uv-python/v0.0.1 \
  git@github.com:lesliebinbin/mise-blueprints.git \
  ./my-python-package
#+end_src

Use =--defaults= to accept every default answer non-interactively:

#+begin_src shell
copier copy \
  --defaults \
  --vcs-ref uv-python/v0.0.1 \
  git@github.com:lesliebinbin/mise-blueprints.git \
  ./my-python-package
#+end_src

* Pinning the template

=--vcs-ref= accepts one Git branch, tag, or commit SHA. For repeatable
automation, use an immutable tag or full commit SHA after the branch is
finalized:

#+begin_src shell
copier copy \
  --vcs-ref <tag-or-full-commit-sha> \
  git@github.com:lesliebinbin/mise-blueprints.git \
  ./my-python-package
#+end_src

* Local development

#+begin_src shell
copier copy \
  --data-file answers.yml.example \
  . \
  /tmp/example-python-package
#+end_src

* Generated package setup

#+begin_src shell
cd my-python-package
mise install
mise setup
mise check
#+end_src
